// Sort on max point impact
                    case "impact":
                        const weight_cat_length = {};
                        for (let w in weights) {
                            weight_cat_length[w] = 0;
                        }
                        for (let g of grades) {
                            if (weight_cat_length[g.weight_cat] != null) {
                                if (Calculate.isNumeric(g.max) && Calculate.isNumeric(g.min)) {
                                    weight_cat_length[g.weight_cat] += 1;
                                }
                            }
                        }

                        grades.sort((a, b) => {
                            const a_is_num = Calculate.isNumeric(a.max);
                            const b_is_num = Calculate.isNumeric(b.max);
                            const a_weight_val = weights[a.weight_cat];
                            const b_weight_val = weights[b.weight_cat];

                            for (let x of [null, ""]) {
                                if (a_weight_val != x && b_weight_val == x) { return 1 * multiplier; }
                                if (a_weight_val == x && b_weight_val != x) { return -1 * multiplier; }
                            }
                            if (a_is_num && !b_is_num) { return 1 * multiplier; }
                            if (!a_is_num && b_is_num) { return -1 * multiplier; }

                            if (a_is_num && b_is_num) {
                                //weight value % * (max points / weight assignments length)
                                const a_analysis = parseFloat(a_weight_val) * (parseFloat(a.max) / weight_cat_length[a.weight_cat]);
                                const b_analysis = parseFloat(b_weight_val) * (parseFloat(b.max) / weight_cat_length[b.weight_cat]);

                                if (a_analysis < b_analysis) { return -1 * multiplier; }
                                if (a_analysis > b_analysis) { return 1 * multiplier; }
                            }
                            return 0;
                        });
                        break;
